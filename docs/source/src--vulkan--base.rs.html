2 //! VulkanBase Module (base.rs)
3 //!
4 //! Created: July 2025  
5 //! Author: Stephen Willey (with the AIs doing a bunch of the work and trying to teach me)
6 //!
7 //! This module defines `VulkanBase`, the one–stop container for all Vulkan state and
8 //! resources in our application. It handles:
9 //!   • Instance creation and destruction  
10 //!   • Physical device selection and logical device + graphics queue setup  
11 //!   • Surface creation via winit + ash_window  
12 //!   • Swapchain management delegated to `Swapchain`  
13 //!   • Graphics pipeline setup delegated to `Pipeline`  
14 //!   • Command pool and command buffer allocation & recording  
15 //!   • Synchronization primitives (semaphores & fence)  
16 //!   
17 //! **Manual Teardown:**  
18 //! Unlike idiomatic Rust RAII, we implement a single `Drop` on `VulkanBase` that
19 //! explicitly destroys every Vulkan object in the correct order—waiting for the GPU
20 //! to idle, then tearing down swapchain, pipeline, sync, command pool, surface, device,
21 //! and finally the instance. This guarantees we never call Vulkan destroy-functions
22 //! on a device or loader that’s already been dropped.
23 //!
24 //! **Usage:**  
25 //! Create a `VulkanBase` via `VulkanBase::new(window, event_loop)`, call
26 //! `draw_frame()` in your render loop, and rely on `Drop` to clean up on exit.
27 //!
28 //! --------------------------------------------------------------------------------------

219     /// Begin a frame: wait/reset fences, acquire the next image, and begin the command buffer.
220     ///
221     /// Returns `Ok(None)` when the swapchain is out-of-date so the caller can skip this frame.
222     ///
223     /// # Steps
224     /// 1. Wait for the previous GPU work that used this CPU frame slot.
225     /// 2. Acquire the next swapchain image (signals the per-slot image-available semaphore).
226     /// 3. If that image is still owned by another slot, wait on that image's owner fence.
227     /// 4. Reset this slot's fence and mark it as the owner of the acquired image.
228     /// 5. Begin the command buffer so the passes can record their own rendering.
229     pub fn begin_frame(&mut self) -> Result<Option<FrameCtx>, Box<dyn Error>> {
230         unsafe {
231             let slot = self.frame_slot;
232 
233             // Wait for the previous work submitted on this CPU slot.
234             self.device
235                 .wait_for_fences(&[self.in_flight_fences[slot]], true, u64::MAX)?;
236 
237             // After the fence is signaled, GPU work for that slot is complete.
238             // If we have a previous image index for this slot, read back its timestamps.
239             if let Some(prev_image_index) = self.last_image_per_slot[slot] {
240                 let base = self.image_query_base(prev_image_index);
241                 let mut data = [0_u64; TIMESTAMPS_PER_IMAGE as usize];
242                 let res = self.device.get_query_pool_results(
243                     self.timestamp_query_pool,
244                     base,
245                     &mut data,
246                     vk::QueryResultFlags::TYPE_64,
247                 );
248                 if res.is_ok() {
249                     self.last_gpu_timings = GpuPassTimings {
250                         total_ms: self.timestamp_delta_ms(data[0], data[4]),
251                         shadow_ms: self.timestamp_delta_ms(data[0], data[1]),
252                         scene_ms: self.timestamp_delta_ms(data[1], data[2]),
253                         vegetation_ms: self.timestamp_delta_ms(data[2], data[3]),
254                     };
255                 }
256                 // Clear the record; we'll set it again on submit
257                 self.last_image_per_slot[slot] = None;
258             }
259 
260             // Acquire an image; signal when it's ready via the per-slot image-available semaphore.
261             let (image_index, is_suboptimal) = match self.swapchain_loader.acquire_next_image(
262                 self.swapchain.handle,
263                 u64::MAX,
264                 self.image_available_semaphores[slot],
265                 vk::Fence::null(),
266             ) {
267                 Ok(result) => result,
268                 Err(vk::Result::ERROR_OUT_OF_DATE_KHR) => {
269                     self.swapchain_recreation_needed = true;
270                     return Ok(None);
271                 }
272                 Err(e) => return Err(e.into()),
273             };
274             self.swapchain_recreation_needed |= is_suboptimal;
275 
276             let idx = image_index as usize;
277             self.current_image_index = idx;
278 
279             // If this image is still tied to an older in-flight slot, wait for that slot to finish first.
280             if self.image_owner_fence[idx] != vk::Fence::null() {
281                 self.device
282                     .wait_for_fences(&[self.image_owner_fence[idx]], true, u64::MAX)?;
283             }
284 
285             // Reuse this slot's fence for the new submit and associate it with this image.
286             self.device.reset_fences(&[self.in_flight_fences[slot]])?;
287             self.image_owner_fence[idx] = self.in_flight_fences[slot];
288 
289             // Begin recording; passes are responsible for starting rendering.
290             let cmd_buf = self.command_buffers[idx];
291 
292             let begin_info = vk::CommandBufferBeginInfo::default();
293             self.device.begin_command_buffer(cmd_buf, &begin_info)?;
294 
295             Ok(Some(FrameCtx {
296                 cmd_buf,
297                 image_index,
298                 frame_slot: slot,
299             }))
300         }
301     }

303     /// Insert image memory barriers for a set of attachment transitions.
304     ///
305     /// Each [`ImageTransition`] describes how a single image's layout and
306     /// access masks change. The command buffer must be in the recording state
307     /// and will receive a single `vkCmdPipelineBarrier` covering all
308     /// transitions.
309     pub fn insert_attachment_barriers(
310         &self,
311         cmd: vk::CommandBuffer,
312         transitions: &[ImageTransition],
313     ) {
314         if transitions.is_empty() {
315             return;
316         }
317         let mut barriers: Vec<vk::ImageMemoryBarrier> = Vec::new();
318         let mut src_stage = vk::PipelineStageFlags::empty();
319         let mut dst_stage = vk::PipelineStageFlags::empty();
320         for t in transitions {
321             barriers.push(vk::ImageMemoryBarrier {
322                 src_access_mask: t.src_access_mask,
323                 dst_access_mask: t.dst_access_mask,
324                 old_layout: t.old_layout,
325                 new_layout: t.new_layout,
326                 src_queue_family_index: vk::QUEUE_FAMILY_IGNORED,
327                 dst_queue_family_index: vk::QUEUE_FAMILY_IGNORED,
328                 image: t.image,
329                 subresource_range: vk::ImageSubresourceRange {
330                     aspect_mask: t.aspect_mask,
331                     base_mip_level: 0,
332                     level_count: 1,
333                     base_array_layer: 0,
334                     layer_count: t.layer_count,
335                 },
336                 ..Default::default()
337             });
338             src_stage |= t.src_stage_mask;
339             dst_stage |= t.dst_stage_mask;
340         }
341         unsafe {
342             self.device.cmd_pipeline_barrier(
343                 cmd,
344                 src_stage,
345                 dst_stage,
346                 vk::DependencyFlags::empty(),
347                 &[],
348                 &[],
349                 &barriers,
350             );
351         }
352     }

354     /// End a frame: end the command buffer, submit, present, and advance the slot.
355     ///
356     /// # Steps
357     /// 1. End the command buffer (each pass has already ended its own rendering).
358     /// 2. Submit the command buffer: wait on the per-slot image-available semaphore and signal the per-image render-finished semaphore.
359     /// 3. Present the image, waiting on the render-finished semaphore for this image.
360     /// 4. Advance to the next CPU frame-in-flight slot.
361     pub fn end_frame(&mut self, frame: FrameCtx) -> Result<(), Box<dyn Error>> {
362         unsafe {
363             // Command buffer was already closed by the passes.
364             self.device.end_command_buffer(frame.cmd_buf)?;
365 
366             // Submit: wait for image-available (slot), signal render-finished (per-image)
367             let wait_stages = [vk::PipelineStageFlags::COLOR_ATTACHMENT_OUTPUT];
368             let wait_sems = [self.image_available_semaphores[frame.frame_slot]];
369             let signal_sems = [self.render_finished_semaphores[frame.image_index as usize]];
370             let submit_info = vk::SubmitInfo {
371                 wait_semaphore_count: 1,
372                 p_wait_semaphores: wait_sems.as_ptr(),
373                 p_wait_dst_stage_mask: wait_stages.as_ptr(),
374                 command_buffer_count: 1,
375                 p_command_buffers: &frame.cmd_buf,
376                 signal_semaphore_count: 1,
377                 p_signal_semaphores: signal_sems.as_ptr(),
378                 ..Default::default()
379             };
380             self.device.queue_submit(
381                 self.graphics_queue,
382                 &[submit_info],
383                 self.in_flight_fences[frame.frame_slot],
384             )?;
385 
386             // Present the image, waiting on the render-finished semaphore for this image
387             let present_wait = [self.render_finished_semaphores[frame.image_index as usize]];
388             let image_index = frame.image_index; // keep a local so we can take a stable pointer
389             let present_info = vk::PresentInfoKHR {
390                 wait_semaphore_count: 1,
391                 p_wait_semaphores: present_wait.as_ptr(),
392                 swapchain_count: 1,
393                 p_swapchains: &self.swapchain.handle,
394                 p_image_indices: &image_index,
395                 ..Default::default()
396             };
397             match self
398                 .swapchain_loader
399                 .queue_present(self.graphics_queue, &present_info)
400             {
401                 Ok(true) | Err(vk::Result::ERROR_OUT_OF_DATE_KHR) => {
402                     self.swapchain_recreation_needed = true;
403                 }
404                 Err(e) => return Err(e.into()),
405                 _ => {}
406             }
407 
408             // Advance to next CPU slot
409             self.frame_slot = (frame.frame_slot + 1) % self.image_available_semaphores.len();
410 
411             // Record which image this slot just submitted, so on next begin_frame we can read timestamps
412             if self.last_image_per_slot.len() != self.image_available_semaphores.len() {
413                 self.last_image_per_slot = vec![None; self.image_available_semaphores.len()];
414             }
415             self.last_image_per_slot[frame.frame_slot] = Some(frame.image_index);
416         }
417         Ok(())
418     }

431     /// Starts the frame's timestamp range and clears its per-pass boundaries.
432     ///
433     /// Queries belong to the acquired swapchain image, so an image is never reset while the
434     /// fence protecting that image still has GPU work in flight.
435     pub fn begin_gpu_timing(&self, cmd: vk::CommandBuffer, image_index: u32) {
436         unsafe {
437             let base = self.image_query_base(image_index);
438             self.device.cmd_reset_query_pool(
439                 cmd,
440                 self.timestamp_query_pool,
441                 base,
442                 TIMESTAMPS_PER_IMAGE,
443             );
444             self.device.cmd_write_timestamp(
445                 cmd,
446                 vk::PipelineStageFlags::TOP_OF_PIPE,
447                 self.timestamp_query_pool,
448                 base + GpuTimestamp::FrameStart as u32,
449             );
450         }
451     }

494     /// Converts a monotonic timestamp pair into milliseconds.  An unavailable or invalid pair
495     /// remains `None` instead of producing a misleading profiler value.
496     fn timestamp_delta_ms(&self, start: u64, end: u64) -> Option<f32> {
497         (end > start).then(|| {
498             let delta_ticks = end - start;
499             let ns = (delta_ticks as f64) * (self.timestamp_period_ns as f64);
500             (ns / 1_000_000.0) as f32
501         })
502     }

504     /// Update the global UBO with lighting data
505     pub fn update_global_ubo(&mut self, image_index: usize, ubo: &GlobalUbo) {
506         let allocator = self.allocator.as_ref().expect("allocator");
507         let allocation = &self.ubo_allocations[image_index];
508         unsafe {
509             // These allocations were created with MAPPED, so VMA keeps the pointer valid
510             // for their lifetime and the per-frame path does not need map/unmap calls.
511             let ptr = allocator.get_allocation_info(allocation).mapped_data as *mut u8;
512             debug_assert!(!ptr.is_null());
513             std::ptr::copy_nonoverlapping(
514                 ubo as *const GlobalUbo as *const u8,
515                 ptr,
516                 std::mem::size_of::<GlobalUbo>(),
517             );
518             allocator
519                 .flush_allocation(allocation, 0, std::mem::size_of::<GlobalUbo>() as u64)
520                 .expect("flush UBO");
521         }
522     }

524     /// Update set=0 binding=1 to point at the current frame's shadow image view
525     pub fn update_shadow_descriptor(&mut self, image_index: usize, image_view: vk::ImageView) {
526         // The view normally stays the same for an image, so avoid a redundant descriptor write.
527         if self.shadow_descriptor_views[image_index] == image_view {
528             return;
529         }
530         let set = self.set0_descriptor_sets[image_index];
531         let image_info = vk::DescriptorImageInfo {
532             sampler: self.shadow_sampler,
533             image_view,
534             image_layout: vk::ImageLayout::SHADER_READ_ONLY_OPTIMAL,
535         };
536         let write = vk::WriteDescriptorSet {
537             dst_set: set,
538             dst_binding: 1,
539             dst_array_element: 0,
540             descriptor_count: 1,
541             descriptor_type: vk::DescriptorType::COMBINED_IMAGE_SAMPLER,
542             p_image_info: &image_info,
543             ..Default::default()
544         };
545         unsafe { self.device.update_descriptor_sets(&[write], &[]) };
546         self.shadow_descriptor_views[image_index] = image_view;
547     }

549     /// Apply any queued surface-dependent changes (like MSAA) at a frame boundary.
550     pub fn apply_pending_surface_changes(
551         &mut self,
552         window: &winit::window::Window,
553     ) -> Result<bool, Box<dyn std::error::Error>> {
554         let mut recreated = false;
555         if let Some(new_samples) = self.pending_msaa_samples.take()
556             && new_samples != self.engine_settings.msaa_samples
557         {
558             self.engine_settings.msaa_samples = new_samples;
559             // Sample count is baked into graphics pipelines; extent is not.
560             self.pipeline_generation = self.pipeline_generation.saturating_add(1);
561             self.recreate_swapchain(window)?;
562             recreated = true;
563         }
564         Ok(recreated)
565     }

571     /// # Returns
572     /// * `Result<Instance, Box<dyn Error>>` - The created Vulkan instance on success, or an error on failure.
573     /// * `Result<(), Box<dyn Error>>` - Returns Ok on success, or an error if the swapchain could not be recreated.
574     pub fn recreate_swapchain(&mut self, window: &Window) -> Result<(), Box<dyn Error>> {
575         let old_color_format = self.swapchain.color_format;
576         let old_depth_format = self.swapchain.depth_format;
577         self.swapchain.recreate(
578             &self.instance,
579             &self.device,
580             self.physical_device,
581             &self.surface,
582             &self.surface_loader,
583             window,
584             self.allocator.as_ref().unwrap(),
585             self.engine_settings.msaa_samples,
586         )?;
587         unsafe {
588             self.device
589                 .free_command_buffers(self.command_pool, &self.command_buffers);
590         }
591         self.command_buffers.clear();
592         self.command_buffers = Self::allocate_command_buffers(
593             &self.device,
594             self.command_pool,
595             vk::CommandBufferLevel::PRIMARY,
596             self.swapchain.swapchain_image_views.len(),
597         )?;
598         // Destroy old per-image present semaphores
599         for sem in self.render_finished_semaphores.drain(..) {
600             unsafe {
601                 self.device.destroy_semaphore(sem, None);
602             }
603         }
604         // Recreate to match new image count
605         let new_image_count = self.swapchain.swapchain_image_views.len();
606         let semaphore_info = vk::SemaphoreCreateInfo::default();
607         self.render_finished_semaphores = Vec::with_capacity(new_image_count);
608         for _ in 0..new_image_count {
609             unsafe {
610                 self.render_finished_semaphores
611                     .push(self.device.create_semaphore(&semaphore_info, None)?);
612             }
613         }
614         // Reset per-image fence tracking
615         self.image_owner_fence = vec![vk::Fence::null(); new_image_count];
616         // Clear ownership as resources are destroyed, so a later allocation
617         // failure cannot make Drop destroy the same handles again.
618         unsafe {
619             self.device
620                 .destroy_descriptor_pool(self.set0_descriptor_pool, None);
621         }
622         self.set0_descriptor_pool = vk::DescriptorPool::null();
623         self.set0_descriptor_sets.clear();
624         for (buffer, mut allocation) in self
625             .ubo_buffers
626             .drain(..)
627             .zip(self.ubo_allocations.drain(..))
628         {
629             unsafe {
630                 self.allocator
631                     .as_ref()
632                     .unwrap()
633                     .destroy_buffer(buffer, &mut allocation)
634             };
635         }
636 
637         (self.ubo_buffers, self.ubo_allocations) =
638             Self::create_uniform_buffers(self.allocator.as_ref().unwrap(), new_image_count)?;
639         (self.set0_descriptor_pool, self.set0_descriptor_sets) =
640             Self::create_set0_descriptor_pool_and_sets(
641                 &self.device,
642                 self.set0_global_layout,
643                 &self.ubo_buffers,
644             )?;
645         self.shadow_descriptor_views = vec![vk::ImageView::null(); new_image_count];
646 
647         self.attachment_manager
648             .cleanup(&self.device, self.allocator.as_ref().unwrap());
649         self.attachment_manager = AttachmentManager::new(new_image_count);
650         // Recreate the timestamp query pool to match new image count and per-pass markers.
651         unsafe {
652             self.device
653                 .destroy_query_pool(self.timestamp_query_pool, None);
654         }
655         self.timestamp_query_pool = vk::QueryPool::null();
656         let qp_info = vk::QueryPoolCreateInfo {
657             query_type: vk::QueryType::TIMESTAMP,
658             query_count: (new_image_count as u32) * TIMESTAMPS_PER_IMAGE,
659             ..Default::default()
660         };
661         self.timestamp_query_pool = unsafe { self.device.create_query_pool(&qp_info, None)? };
662         // Invalidate last GPU time reading since the pool was recreated
663         self.last_gpu_timings = GpuPassTimings::default();
664         // Avoid reading uninitialized queries on the next frame after recreation
665         self.last_image_per_slot = vec![None; self.image_available_semaphores.len()];
666         // Dynamic viewport/scissor removed extent from pipeline compatibility. A rare
667         // surface-format change still requires every graphics pipeline to be rebuilt.
668         if self.swapchain.color_format != old_color_format
669             || self.swapchain.depth_format != old_depth_format
670         {
671             self.pipeline_generation = self.pipeline_generation.saturating_add(1);
672         }
673         self.swapchain_recreation_needed = false;
674         Ok(())
675     }

677     const MSAA_CHOICES: [u32; 7] = [1, 2, 4, 8, 16, 32, 64];
678 
679     fn flag_for(samples: u32) -> vk::SampleCountFlags {
680         vk::SampleCountFlags::from_raw(samples)
681     }
682 
683     fn clamp_msaa_samples(&self, desired: u32) -> u32 {
684         let mut opts = self.supported_msaa_samples();
685         if opts.is_empty() {
686             return 1;
687         }
688         opts.sort_unstable();
689         // highest supported <= desired, else the smallest supported
690         opts.iter()
691             .copied()
692             .rev()
693             .find(|&n| n <= desired)
694             .unwrap_or(opts[0])
695     }
696 
697     /// Returns the list of supported MSAA sample counts for the current device.
698     pub fn supported_msaa_samples(&self) -> Vec<u32> {
699         let supported = self.sample_count_flags_supported;
700         Self::MSAA_CHOICES
701             .into_iter()
702             .filter(|&n| supported.contains(Self::flag_for(n)))
703             .collect()
704     }
705 
706     /// Requests a new MSAA sample count. The swapchain/pipelines will be
707     /// recreated on the next safe point to apply the change.
708     pub fn request_msaa_samples(&mut self, desired: u32) {
709         self.pending_msaa_samples = Some(self.clamp_msaa_samples(desired));
710     }

712     /// Current MSAA sample count in effect.
713     pub fn get_msaa_samples(&self) -> u32 {
714         self.engine_settings.msaa_samples
715     }

717     /// Toggles the wireframe mode in the debug settings.
718     pub fn toggle_wireframe(&mut self) {
719         if !self.supports_wireframe {
720             return;
721         }
722         let old = self.engine_settings.wireframe;
723         self.engine_settings.wireframe = !self.engine_settings.wireframe;
724         if self.engine_settings.wireframe != old {
725             // Wireframe affects pipelines; ensure rebuild next frame
726             self.pipeline_generation = self.pipeline_generation.saturating_add(1);
727         }
728     }
729     /// Returns the current pipeline generation counter.
730     pub fn pipeline_generation(&self) -> u64 {
731         self.pipeline_generation
732     }

734     /// Returns and clears the pending swapchain-recreation request.
735     pub fn take_swapchain_recreation_request(&mut self) -> bool {
736         std::mem::take(&mut self.swapchain_recreation_needed)
737     }

739     /// Queue a resize for the next frame boundary. Several window events collapse into one.
740     pub fn request_swapchain_recreation(&mut self) {
741         self.swapchain_recreation_needed = true;
742     }

745 impl Drop for VulkanBase {
746     /// Cleans up Vulkan resources when the `VulkanBase` is dropped.
747     /// This includes waiting for the device to be idle,
748     /// cleaning up the swapchain, pipeline, command pool,
749     /// and other Vulkan objects.
750     fn drop(&mut self) {
751         println!("💧 Dropping VulkanBase");
752         unsafe {
753             // `Drop` gives us `&mut self`, but Vulkan does not know Rust's
754             // ownership graph.  Destroy resources in reverse dependency order.
755             // Destructors must not panic, including after device loss or while
756             // another error is unwinding the application's ownership tree.
757             if let Err(error) = self.device.device_wait_idle() {
758                 eprintln!("Failed to wait for device during shutdown: {error}");
759             }
760 
761             self.attachment_manager
762                 .cleanup(&self.device, self.allocator.as_ref().unwrap());
763             self.swapchain.cleanup(
764                 &self.instance,
765                 &self.device,
766                 self.allocator.as_ref().unwrap(),
767             );
768             self.device
769                 .destroy_descriptor_set_layout(self.set0_global_layout, None);
770 
771             // Then destroy the rest of the resources
772             for &sem in &self.image_available_semaphores {
773                 self.device.destroy_semaphore(sem, None);
774             }
775             for &sem in &self.render_finished_semaphores {
776                 self.device.destroy_semaphore(sem, None);
777             }
778             for &fence in &self.in_flight_fences {
779                 self.device.destroy_fence(fence, None);
780             }
781             self.device
782                 .destroy_query_pool(self.timestamp_query_pool, None);
783             for &buffer in &self.command_buffers {
784                 self.device
785                     .free_command_buffers(self.command_pool, &[buffer]);
786             }
787             // UBO and descriptor resources
788             for (buf, alloc) in self.ubo_buffers.iter().zip(self.ubo_allocations.iter_mut()) {
789                 self.allocator.as_ref().unwrap().destroy_buffer(*buf, alloc);
790             }
791             self.device
792                 .destroy_descriptor_pool(self.set0_descriptor_pool, None);
793             self.device.destroy_sampler(self.shadow_sampler, None);
794             self.device
795                 .destroy_pipeline_cache(self.pipeline_cache, None);
796             self.device.destroy_command_pool(self.command_pool, None);
797             // `Option::take` moves the allocator out while we only have `&mut self`.
798             // Dropping it here frees VMA memory before its Vulkan device disappears.
799             if let Some(alloc) = self.allocator.take() {
800                 // (optional) quick summary before it goes away
801                 if let Ok(stats) = alloc.calculate_statistics() {
802                     let s = stats.total.statistics;
803                     println!(
804                         "🗑️ VMA total before drop: allocs={} blocks={} allocBytes={} blockBytes={}",
805                         s.allocationCount, s.blockCount, s.allocationBytes, s.blockBytes
806                     );
807                 }
808                 drop(alloc);
809             }
810             self.device.destroy_device(None);
811             self.surface_loader.destroy_surface(self.surface, None);
812             #[cfg(debug_assertions)]
813             self.debug_utils_loader
814                 .destroy_debug_utils_messenger(self.debug_messenger, None);
815             self.instance.destroy_instance(None);
816         }
817     }
818 }
