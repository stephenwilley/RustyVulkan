18 /// Render pass responsible for drawing the ImGui user interface.
19 pub struct UiPass {
20     attachments: [AttachmentRequest; 2],
21     show_point_lights_window: bool,
22     show_sun_window: bool,
23     show_ground_window: bool,
24     show_shadow_map_window: bool,
25     show_profiler_window: bool,
26 }
27 
28 impl UiPass {
29     /// Create a new [`UiPass`].
30     pub fn new() -> Self {
31         Self {
32             attachments: [
33                 AttachmentRequest::new(AttachmentKind::SwapchainColor),
34                 // The cascade preview samples the same shadow map as the main pass.
35                 AttachmentRequest::new(AttachmentKind::Shadow),
36             ],
37             show_point_lights_window: false,
38             show_sun_window: false,
39             show_ground_window: false,
40             show_shadow_map_window: false,
41             show_profiler_window: false,
42         }
43     }
44 }
45 
46 impl RenderPass for UiPass {
47     /// Render the ImGui user interface over the final swapchain image.
48     fn execute(&mut self, ctx: &mut RenderCtx) -> Result<(), Box<dyn std::error::Error>> {
49         let cmd = ctx.frame.cmd_buf;
50         let ui_ctx = ctx.ui_ctx.as_mut().unwrap();
51 
52         ui_ctx
53             .platform
54             .prepare_frame(ui_ctx.imgui.io_mut(), ui_ctx.window)
55             .expect("Failed to prepare imgui frame");
56 
57         let ui = ui_ctx.imgui.frame();
58         if ui_ctx.show_ui
59             && let Some(_menu_bar) = ui.begin_main_menu_bar()
60         {
61             if let Some(_main_menu) = ui.begin_menu("Main") {
62                 if ui.menu_item("Exit") {
63                     ui_ctx.exit_flag.store(true, Ordering::Relaxed);
64                 }
65                 _main_menu.end();
66             }
67 
68             if let Some(_world_menu) = ui.begin_menu("World Controls") {
69                 if ui.menu_item("Point Lights") {
70                     self.show_point_lights_window = true;
71                 }
72                 if ui.menu_item("Sun Light") {
73                     self.show_sun_window = true;
74                 }
75                 if ui.menu_item("Ground Surface") {
76                     self.show_ground_window = true;
77                 }
78                 _world_menu.end();
79             }
80 
81             if let Some(_engine_menu) = ui.begin_menu("Engine") {
82                 if let Some(_msaa_menu) = ui.begin_menu("MSAA") {
83                     let options = ctx.vulkan_base.supported_msaa_samples();
84                     let mut current = ctx.vulkan_base.get_msaa_samples() as i32;
85                     let prev = current;
86                     for (i, samples) in options.iter().enumerate() {
87                         ui.radio_button(format!("{}x", samples), &mut current, *samples as i32);
88                         if i + 1 != options.len() {
89                             ui.same_line();
90                         }
91                     }
92                     ui.new_line();
93                     if current != prev {
94                         ctx.vulkan_base.request_msaa_samples(current as u32)
95                     }
96                     _msaa_menu.end();
97                 }
98                 if let Some(_shadow_menu) = ui.begin_menu("Cascade Resolution") {
99                     let mut cur_res = ctx.vulkan_base.engine_settings.shadow_map_resolution as i32;
100                     let prev = cur_res;
101                     let choices = [512, 1024, 2048, 4096];
102                     for (i, &opt) in choices.iter().enumerate() {
103                         ui.radio_button(format!("{}", opt), &mut cur_res, opt);
104                         if i + 1 != choices.len() {
105                             ui.same_line();
106                         }
107                     }
108                     ui.new_line();
109                     if cur_res != prev {
110                         ctx.vulkan_base.engine_settings.shadow_map_resolution = cur_res as u32;
111                     }
112                     _shadow_menu.end();
113                 }
114                 // Checkable toggle: wireframe on/off
115                 let wireframe = ctx.vulkan_base.engine_settings.wireframe;
116                 if ui
117                     .menu_item_config("Wireframe")
118                     .selected(wireframe)
119                     .enabled(ctx.vulkan_base.supports_wireframe)
120                     .build()
121                 {
122                     ctx.vulkan_base.toggle_wireframe();
123                 }
124                 if ui.menu_item("Show Shadow Map") {
125                     self.show_shadow_map_window = true;
126                 }
127                 if ui.menu_item("Profiler") {
128                     self.show_profiler_window = true;
129                 }
130                 _engine_menu.end();
131             }
132 
133             // Right-align the text with CPU and optional GPU time
134             let text = match ui_ctx.gpu_ms_per_frame {
135                 Some(g) => format!("CPU: {:.2} ms | GPU: {:.2} ms", ui_ctx.cpu_ms_per_frame, g),
136                 None => format!("CPU: {:.2} ms | GPU: --", ui_ctx.cpu_ms_per_frame),
137             };
138             let text_size = ui.calc_text_size(&text);
139             let menu_bar_width = ui.content_region_avail()[0];
140             ui.set_cursor_pos([menu_bar_width - text_size[0], 0.0]);
141             ui.text(text);
142 
143             _menu_bar.end();
144         }
145 
146         if self.show_profiler_window {
147             let mut open = true;
148             ui.window("Profiler")
149                 .opened(&mut open)
150                 .always_auto_resize(true)
151                 .build(|| {
152                     ui.text("Previous completed-frame timings");
153                     ui.separator();
154                     match ui_ctx.gpu_ms_per_frame {
155                         Some(g) => ui.text(format!("Total GPU: {:.2} ms", g)),
156                         None => ui.text("Total GPU: --"),
157                     }
158                     let timings = ui_ctx.gpu_pass_timings;
159                     match timings.shadow_ms {
160                         Some(ms) => ui.text(format!("  Shadow: {:.2} ms", ms)),
161                         None => ui.text("  Shadow: --"),
162                     }
163                     match timings.scene_ms {
164                         Some(ms) => ui.text(format!("  Scene before vegetation: {:.2} ms", ms)),
165                         None => ui.text("  Scene before vegetation: --"),
166                     }
167                     match timings.vegetation_ms {
168                         Some(ms) => ui.text(format!("  Grass + reeds: {:.2} ms", ms)),
169                         None => ui.text("  Grass + reeds: --"),
170                     }
171                     ui.text(format!(
172                         "Total CPU record: {:.2} ms",
173                         ui_ctx.cpu_ms_per_frame
174                     ));
175 
176                     ui.new_line();
177                     ui.text("CPU ms (last ~5s)");
178                     if !ui_ctx.cpu_ms_history.is_empty() {
179                         let cpu_max = ui_ctx
180                             .cpu_ms_history
181                             .iter()
182                             .cloned()
183                             .fold(0.0_f32, f32::max);
184                         let cpu_overlay = format!("max {:.1} ms", cpu_max);
185                         PlotLines::new(ui, "CPU", ui_ctx.cpu_ms_history)
186                             .graph_size([300.0, 80.0])
187                             .scale_min(0.0)
188                             .scale_max(33.0)
189                             .overlay_text(&cpu_overlay)
190                             .build();
191                     } else {
192                         ui.text("(no data yet)");
193                     }
194 
195                     ui.new_line();
196                     ui.text("GPU ms (last ~5s)");
197                     if !ui_ctx.gpu_ms_history.is_empty() {
198                         let gpu_max = ui_ctx
199                             .gpu_ms_history
200                             .iter()
201                             .cloned()
202                             .fold(0.0_f32, f32::max);
203                         let gpu_overlay = format!("max {:.1} ms", gpu_max);
204                         PlotLines::new(ui, "GPU", ui_ctx.gpu_ms_history)
205                             .graph_size([300.0, 80.0])
206                             .scale_min(0.0)
207                             .scale_max(33.0)
208                             .overlay_text(&gpu_overlay)
209                             .build();
210                     } else {
211                         ui.text("(no data yet)");
212                     }
213                 });
214             if !open {
215                 self.show_profiler_window = false;
216             }
217         }
218 
219         // Shadow Map debug window
220         if self.show_shadow_map_window {
221             // Every image/layer pair has a descriptor, so all widgets keep their
222             // own cascade while the swapchain rotates through in-flight images.
223             let shadow_tex_ids: Option<[imgui::TextureId; SHADOW_CASCADE_COUNT]> =
224                 ctx.attachments.get(&AttachmentKind::Shadow).map(|handle| {
225                     std::array::from_fn(|cascade| {
226                         ui_ctx.renderer.shadow_texture_id(
227                             ctx.vulkan_base,
228                             ctx.frame.image_index as usize,
229                             cascade,
230                             ctx.vulkan_base.shadow_sampler,
231                             handle.layer_views[cascade],
232                         )
233                     })
234                 });
235 
236             let mut open = true;
237             ui.window("Shadow Cascades")
238                 .opened(&mut open)
239                 .always_auto_resize(true)
240                 .build(|| {
241                     if let Some(texture_ids) = &shadow_tex_ids {
242                         for (cascade, &texture_id) in texture_ids.iter().enumerate() {
243                             ui.group(|| {
244                                 ui.text(format!("Cascade {}", cascade + 1));
245                                 imgui::Image::new(texture_id, [224.0, 224.0])
246                                     .uv0([0.0, 1.0])
247                                     .uv1([1.0, 0.0])
248                                     .build(ui);
249                             });
250                             if cascade % 2 == 0 {
251                                 ui.same_line();
252                             }
253                         }
254                     } else {
255                         ui.text("Shadow attachment not available");
256                     }
257                 });
258             if !open {
259                 self.show_shadow_map_window = false;
260             }
261         }
262 
263         if self.show_point_lights_window {
264             let mut open = true;
265             ui.window("Point Lights")
266                 .opened(&mut open)
267                 .always_auto_resize(true)
268                 .build(|| {
269                     ui.text("Adjust point light positions");
270                     ui.separator();
271 
272                     let count = ctx
273                         .world_controls
274                         .light_count
275                         .min(crate::app::app::MAX_LIGHTS);
276                     for i in 0..count {
277                         ui.text(format!("Light {}", i + 1));
278                         let l = &mut ctx.world_controls.lights[i];
279 
280                         // X slider: [-5, 5]
281                         ui.slider(format!("X##{}", i), -5.0, 5.0, &mut l.position[0]);
282                         // Y slider: [0, 5] (default 2.0)
283                         ui.slider(format!("Y##{}", i), 0.0, 5.0, &mut l.position[1]);
284                         // Z slider: [-5, 5]
285                         ui.slider(format!("Z##{}", i), -5.0, 5.0, &mut l.position[2]);
286                         // Intensity slider: [0, 5]
287                         ui.slider(format!("Intensity##{}", i), 0.0, 5.0, &mut l.intensity);
288                         // Full-range color editor for RGB
289                         ui.color_edit3(format!("Color##{}", i), &mut l.color);
290 
291                         if i + 1 != count {
292                             ui.separator();
293                         }
294                     }
295                 });
296             if !open {
297                 self.show_point_lights_window = false;
298             }
299         }
300 
301         if self.show_sun_window {
302             let mut open = true;
303             ui.window("Sun Light")
304                 .opened(&mut open)
305                 .always_auto_resize(true)
306                 .build(|| {
307                     ui.text("Adjust directional sun light");
308                     ui.separator();
309 
310                     // Direction sliders in [-1, 1], then normalize
311                     let mut dir = ctx.world_controls.sun_direction;
312                     ui.slider("Dir X", -1.0, 1.0, &mut dir[0]);
313                     ui.slider("Dir Y", -1.0, 0.0, &mut dir[1]);
314                     ui.slider("Dir Z", -1.0, 1.0, &mut dir[2]);
315 
316                     // Normalize to keep it a unit vector
317                     let len = (dir[0] * dir[0] + dir[1] * dir[1] + dir[2] * dir[2]).sqrt();
318                     if len > 1e-6 {
319                         dir[0] /= len;
320                         dir[1] /= len;
321                         dir[2] /= len;
322                     }
323                     ctx.world_controls.sun_direction = dir;
324 
325                     // Intensity and color
326                     ui.slider("Intensity", 0.0, 5.0, &mut ctx.world_controls.sun_intensity);
327                     ui.color_edit3("Color", &mut ctx.world_controls.sun_color);
328                 });
329             if !open {
330                 self.show_sun_window = false;
331             }
332         }
333 
334         if self.show_ground_window {
335             let mut open = true;
336             ui.window("Ground Surface")
337                 .opened(&mut open)
338                 .always_auto_resize(true)
339                 .build(|| {
340                     ui.text("Choose ground rendering");
341                     ui.separator();
342                     let mut use_terrain = ctx.world_controls.use_terrain_ground;
343                     ui.radio_button("Infinite Plane", &mut use_terrain, false);
344                     ui.same_line();
345                     ui.radio_button("Terrain (dirt)", &mut use_terrain, true);
346                     ctx.world_controls.use_terrain_ground = use_terrain;
347                 });
348             if !open {
349                 self.show_ground_window = false;
350             }
351         }
352 
353         // MSAA controlled via Engine Settings submenu
354 
355         // (Wireframe uses checkable menu item; no pop-up window)
356 
357         ui_ctx.platform.prepare_render(ui, ui_ctx.window);
358         let draw_data = ui_ctx.imgui.render();
359 
360         let color_att = ctx.attachments[&AttachmentKind::SwapchainColor];
361 
362         let color_attachment_info = vk::RenderingAttachmentInfo::default()
363             .image_view(color_att.view)
364             .image_layout(vk::ImageLayout::COLOR_ATTACHMENT_OPTIMAL)
365             .load_op(vk::AttachmentLoadOp::LOAD) // Load the results of the main pass
366             .store_op(vk::AttachmentStoreOp::STORE); // Store the UI on top
367 
368         let rendering_info = vk::RenderingInfo::default()
369             .render_area(vk::Rect2D {
370                 offset: vk::Offset2D::default(),
371                 extent: ctx.vulkan_base.swapchain.extent,
372             })
373             .layer_count(1)
374             .color_attachments(std::slice::from_ref(&color_attachment_info));
375 
376         unsafe {
377             ctx.vulkan_base
378                 .device
379                 .cmd_begin_rendering(cmd, &rendering_info);
380 
381             ctx.vulkan_base.device.cmd_bind_pipeline(
382                 cmd,
383                 vk::PipelineBindPoint::GRAPHICS,
384                 ui_ctx.renderer.vk_pipeline,
385             );
386 
387             ui_ctx.renderer.render(
388                 &ctx.vulkan_base.device,
389                 ctx.vulkan_base.allocator.as_ref().unwrap(),
390                 cmd,
391                 draw_data,
392                 ctx.frame.image_index as usize,
393             );
394 
395             ctx.vulkan_base.device.cmd_end_rendering(cmd);
396         }
397 
398         Ok(())
399     }
400 
401     fn attachments(&self) -> &[AttachmentRequest] {
402         &self.attachments
403     }
404 
405     fn attachment_info(&self, kind: AttachmentKind) -> (vk::ImageLayout, vk::AccessFlags) {
406         match kind {
407             // LOAD and blending read the main pass's output, so declaring the read makes
408             // the graph insert a barrier after the main pass's writes.
409             AttachmentKind::SwapchainColor => (
410                 vk::ImageLayout::COLOR_ATTACHMENT_OPTIMAL,
411                 vk::AccessFlags::COLOR_ATTACHMENT_READ | vk::AccessFlags::COLOR_ATTACHMENT_WRITE,
412             ),
413             AttachmentKind::Shadow => (
414                 vk::ImageLayout::SHADER_READ_ONLY_OPTIMAL,
415                 vk::AccessFlags::SHADER_READ,
416             ),
417             _ => (vk::ImageLayout::UNDEFINED, vk::AccessFlags::empty()),
418         }
419     }
