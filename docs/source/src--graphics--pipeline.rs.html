2 //! Pipeline Module (pipeline.rs)
3 //!
4 //! Created: July 2025  
5 //! Author: Stephen Willey (with the AIs doing a bunch of the work and trying to teach me)
6 //!
7 //! This module defines the `Pipeline` type, which encapsulates creation and management of
8 //! the Vulkan graphics pipeline. It handles:
9 //!   • Loading SPIR-V shader stages (vertex & fragment)  
10 //!   • Configuring fixed-function state (vertex input, input assembly, viewport & scissor
11 //!     rasterizer, multisampling, color blending)  
12 //!   • Creating the `vk::PipelineLayout` and `vk::Pipeline` objects  
13 //!   • Providing a `cleanup` method to destroy the pipeline layout when no longer needed  
14 //!
15 //! Usage:
16 //!   1. `Pipeline::new`            - initializes the pipeline layout  
17 //!   2. `create_graphics_pipeline` - stitches together all shader and fixed-function state into a usable pipeline  
18 //!   3. `cleanup`                  - tears down the pipeline layout  
19 //!
20 //! This keeps all graphics-pipeline logic centralized and reusable by `VulkanBase`.
21 //! --------------------------------------------------------------------------------------

34 /// Represents the Vulkan graphics pipeline, including shader modules and layout.
35 /// It encapsulates the shader modules used for vertex and fragment stages,
36 /// and the pipeline layout used for rendering.
37 /// It is responsible for creating the graphics pipeline and managing its resources.
38 /// This struct is used by the `VulkanBase` to set up the rendering pipeline.
39 pub struct Pipeline {
40     pub vk_layout: vk::PipelineLayout,
41     pub vk_pipeline: vk::Pipeline,
42     pub depth_write: bool,
43     /// Opaque pipelines avoid the blending read/modify/write path.
44     pub alpha_blending: bool,
45     /// Background passes do not need the depth buffer at all.
46     depth_test: bool,
47     /// Debug wireframe applies to scene geometry, but should not turn the sky into one triangle.
48     honor_wireframe: bool,
49 }

52     /// Creates a pipeline layout with alpha blending enabled, preserving the original material
53     /// pipeline behaviour.  Shader modules are supplied later when the graphics pipeline is built.
54     /// # Arguments
55     /// * `device` - The Vulkan logical device to use for creating the pipeline.
56     /// * `set_layouts` - The descriptor set layouts to use for the pipeline.
57     /// * `depth_write` - Whether depth writing is enabled for this pipeline.
58     /// # Returns
59     /// * `Result<Self, Box<dyn Error>>` - Returns the initialized `Pipeline` on success, or an error on failure.
60     /// # Errors
61     /// * Returns an error if the pipeline layout cannot be created.
62     pub fn new(
63         device: &ash::Device,
64         set_layouts: &[vk::DescriptorSetLayout],
65         depth_write: bool,
66     ) -> Result<Self, Box<dyn Error>> {
67         Self::new_with_options(device, set_layouts, depth_write, true, true, true)
68     }

88     fn new_with_options(
89         device: &ash::Device,
90         set_layouts: &[vk::DescriptorSetLayout],
91         depth_write: bool,
92         alpha_blending: bool,
93         depth_test: bool,
94         honor_wireframe: bool,
95     ) -> Result<Self, Box<dyn Error>> {
96         // 1 - Define a PushConstantRange for two mat4 (mvp, mv) plus a vec4 for UV tiling
97         // (xy used, zw padding): 144 bytes, matching the `Push` block in main.vert.
98         let push_constant_range = vk::PushConstantRange {
99             stage_flags: vk::ShaderStageFlags::VERTEX | vk::ShaderStageFlags::FRAGMENT,
100             offset: 0,
101             size: PUSH_CONSTANT_BYTES,
102         };
103 
104         // 2 - Build your PipelineLayoutCreateInfo with that push-constant baked in and descriptor set layouts
105         let layout_info = vk::PipelineLayoutCreateInfo {
106             set_layout_count: set_layouts.len() as u32,
107             p_set_layouts: set_layouts.as_ptr(),
108             push_constant_range_count: 1,
109             p_push_constant_ranges: &push_constant_range as *const _,
110             ..Default::default()
111         };
112 
113         // 3 - Create the layout as before
114         let vk_layout = unsafe { device.create_pipeline_layout(&layout_info, None)? };
115         println!("🛠️ Pipeline layout created with push‐constant support");
116 
117         Ok(Self {
118             vk_layout,
119             vk_pipeline: vk::Pipeline::null(),
120             depth_write,
121             alpha_blending,
122             depth_test,
123             honor_wireframe,
124         })
125     }

127     /// Loads shaders, ties them to the given formats/extent, and creates the pipeline.
128     /// # Arguments
129     /// * `device` - The Vulkan logical device.
130     /// * `pipeline_cache` - Shared cache that reuses pipeline compilation work.
131     /// * `extent` - The extent of the swapchain.
132     /// * `color_format` - The color attachment format.
133     /// * `depth_format` - The depth attachment format.
134     /// * `shader_infos` - The shader stage information.
135     /// * `wireframe` - Whether to enable wireframe mode.
136     /// # Returns
137     /// * `Result<(), Box<dyn Error>>` - Returns Ok on success, or an error on failure.
138     // The parameters are independent pieces of Vulkan pipeline state.
139     #[allow(clippy::too_many_arguments)]
140     pub fn create_graphics_pipeline(
141         &mut self,
142         device: &ash::Device,
143         pipeline_cache: vk::PipelineCache,
144         extent: vk::Extent2D,
145         color_format: vk::Format,
146         depth_format: vk::Format,
147         shader_infos: &[&ShaderStageInfo],
148         engine_settings: &EngineSettings,
149     ) -> Result<(), Box<dyn Error>> {
150         let binding_descs = [Vertex::binding_description()];
151         let attribute_descs = Vertex::attribute_descriptions();
152         self.create_graphics_pipeline_with_vertex_input(
153             device,
154             pipeline_cache,
155             extent,
156             color_format,
157             depth_format,
158             shader_infos,
159             engine_settings,
160             &binding_descs,
161             &attribute_descs,
162             vk::CullModeFlags::BACK,
163         )
164     }

166     /// Creates a graphics pipeline with caller-provided vertex and instance bindings.
167     ///
168     /// Most materials use [`Vertex`]'s single binding through
169     /// [`Self::create_graphics_pipeline`].  Instanced renderers such as grass add a
170     /// second, per-instance binding while sharing all the usual Vulkan state.
171     #[allow(clippy::too_many_arguments)]
172     pub fn create_graphics_pipeline_with_vertex_input(
173         &mut self,
174         device: &ash::Device,
175         pipeline_cache: vk::PipelineCache,
176         _extent: vk::Extent2D,
177         color_format: vk::Format,
178         depth_format: vk::Format,
179         shader_infos: &[&ShaderStageInfo],
180         engine_settings: &EngineSettings,
181         binding_descs: &[vk::VertexInputBindingDescription],
182         attribute_descs: &[vk::VertexInputAttributeDescription],
183         cull_mode: vk::CullModeFlags,
184     ) -> Result<(), Box<dyn Error>> {
185         let vertex_input_info = vk::PipelineVertexInputStateCreateInfo {
186             vertex_binding_description_count: binding_descs.len() as u32,
187             p_vertex_binding_descriptions: binding_descs.as_ptr(),
188             vertex_attribute_description_count: attribute_descs.len() as u32,
189             p_vertex_attribute_descriptions: attribute_descs.as_ptr(),
190             ..Default::default()
191         };
192 
193         let input_assembly = vk::PipelineInputAssemblyStateCreateInfo {
194             topology: vk::PrimitiveTopology::TRIANGLE_LIST,
195             primitive_restart_enable: vk::FALSE,
196             ..Default::default()
197         };
198         // Viewport and scissor depend on the window size, not on shader compatibility.
199         // Making them dynamic lets a resized swapchain keep the existing pipelines.
200         let viewport_state = vk::PipelineViewportStateCreateInfo {
201             viewport_count: 1,
202             scissor_count: 1,
203             ..Default::default()
204         };
205         let dynamic_states = [vk::DynamicState::VIEWPORT, vk::DynamicState::SCISSOR];
206         let dynamic_state = vk::PipelineDynamicStateCreateInfo {
207             dynamic_state_count: dynamic_states.len() as u32,
208             p_dynamic_states: dynamic_states.as_ptr(),
209             ..Default::default()
210         };
211 
212         let rasterizer = vk::PipelineRasterizationStateCreateInfo {
213             depth_clamp_enable: vk::FALSE,
214             rasterizer_discard_enable: vk::FALSE,
215             polygon_mode: if self.honor_wireframe && engine_settings.wireframe {
216                 vk::PolygonMode::LINE
217             } else {
218                 vk::PolygonMode::FILL
219             },
220             line_width: 1.0,
221             cull_mode,
222             front_face: vk::FrontFace::COUNTER_CLOCKWISE,
223             depth_bias_enable: vk::FALSE,
224             ..Default::default()
225         };
226         let multisampling = vk::PipelineMultisampleStateCreateInfo {
227             rasterization_samples: vk::SampleCountFlags::from_raw(engine_settings.msaa_samples),
228             sample_shading_enable: vk::FALSE,
229             ..Default::default()
230         };
231 
232         // Most imported materials use alpha blending; opaque specialised geometry can skip it.
233         let color_blend_attachment = vk::PipelineColorBlendAttachmentState {
234             color_write_mask: vk::ColorComponentFlags::R
235                 | vk::ColorComponentFlags::G
236                 | vk::ColorComponentFlags::B
237                 | vk::ColorComponentFlags::A,
238             blend_enable: if self.alpha_blending {
239                 vk::TRUE
240             } else {
241                 vk::FALSE
242             },
243             src_color_blend_factor: vk::BlendFactor::SRC_ALPHA,
244             dst_color_blend_factor: vk::BlendFactor::ONE_MINUS_SRC_ALPHA,
245             color_blend_op: vk::BlendOp::ADD,
246             src_alpha_blend_factor: vk::BlendFactor::ONE,
247             dst_alpha_blend_factor: vk::BlendFactor::ZERO,
248             alpha_blend_op: vk::BlendOp::ADD,
249         };
250 
251         let color_blending = vk::PipelineColorBlendStateCreateInfo {
252             logic_op_enable: vk::FALSE,
253             attachment_count: 1,
254             p_attachments: &color_blend_attachment,
255             ..Default::default()
256         };
257 
258         let shader_stage_create_infos: Vec<_> =
259             shader_infos.iter().map(|s| s.to_create_info()).collect();
260 
261         let depth_stencil = vk::PipelineDepthStencilStateCreateInfo {
262             depth_test_enable: if self.depth_test { vk::TRUE } else { vk::FALSE },
263             depth_write_enable: if self.depth_write {
264                 vk::TRUE
265             } else {
266                 vk::FALSE
267             },
268             depth_compare_op: vk::CompareOp::LESS,
269             // stencil is off for now—
270             stencil_test_enable: vk::FALSE,
271             front: Default::default(),
272             back: Default::default(),
273             ..Default::default()
274         };
275 
276         let color_formats = [color_format];
277         let rendering_info = vk::PipelineRenderingCreateInfo {
278             color_attachment_count: color_formats.len() as u32,
279             p_color_attachment_formats: color_formats.as_ptr(),
280             depth_attachment_format: depth_format,
281             ..Default::default()
282         };
283 
284         let mut pipeline_info = vk::GraphicsPipelineCreateInfo {
285             stage_count: shader_stage_create_infos.len() as u32,
286             p_stages: shader_stage_create_infos.as_ptr(),
287             p_vertex_input_state: &vertex_input_info,
288             p_input_assembly_state: &input_assembly,
289             p_viewport_state: &viewport_state,
290             p_rasterization_state: &rasterizer,
291             p_multisample_state: &multisampling,
292             p_color_blend_state: &color_blending,
293             p_depth_stencil_state: &depth_stencil,
294             p_dynamic_state: &dynamic_state,
295             layout: self.vk_layout,
296             render_pass: vk::RenderPass::null(),
297             subpass: 0,
298             ..Default::default()
299         };
300         pipeline_info.p_next = &rendering_info as *const _ as *const std::ffi::c_void;
301 
302         let pipelines = unsafe {
303             device
304                 .create_graphics_pipelines(pipeline_cache, &[pipeline_info], None)
305                 .map_err(|(_, e)| e)?
306         };
307 
308         println!(
309             "🛠️ Graphics pipeline created with {} stages",
310             pipelines.len()
311         );
312         self.vk_pipeline = pipelines[0];
313         Ok(())
314     }

316     /// Recreate the pipeline
317     /// # Arguments
318     /// * `device` - The Vulkan logical device.
319     /// * `pipeline_cache` - Shared cache that reuses pipeline compilation work.
320     /// * `extent` - The extent of the swapchain.
321     /// * `color_format` - The color attachment format.
322     /// * `depth_format` - The depth attachment format.
323     /// * `shader_infos` - The shader stage information.
324     /// * `wireframe` - Whether to enable wireframe mode.
325     /// # Returns
326     /// * `Result<(), Box<dyn Error>>` - Returns Ok on success, or an error on failure.
327     #[allow(clippy::too_many_arguments)]
328     pub fn recreate(
329         &mut self,
330         device: &ash::Device,
331         pipeline_cache: vk::PipelineCache,
332         extent: vk::Extent2D,
333         color_format: vk::Format,
334         depth_format: vk::Format,
335         shader_infos: &[&ShaderStageInfo],
336         engine_settings: &EngineSettings,
337     ) -> Result<(), Box<dyn Error>> {
338         let old_pipeline = self.vk_pipeline;
339         self.create_graphics_pipeline(
340             device,
341             pipeline_cache,
342             extent,
343             color_format,
344             depth_format,
345             shader_infos,
346             engine_settings,
347         )?;
348         // Keep the old pipeline alive until its replacement exists. If creation fails,
349         // `?` returns while the still-valid old handle remains owned by `self`.
350         unsafe {
351             device.destroy_pipeline(old_pipeline, None);
352         }
353         Ok(())
354     }

358     pub fn recreate_with_vertex_input(

366         engine_settings: &EngineSettings,

368         attribute_descs: &[vk::VertexInputAttributeDescription],
369         cull_mode: vk::CullModeFlags,
370     ) -> Result<(), Box<dyn Error>> {
371         let old_pipeline = self.vk_pipeline;
372         self.create_graphics_pipeline_with_vertex_input(
373             device,
374             pipeline_cache,
375             extent,
376             color_format,
377             depth_format,
378             shader_infos,
379             engine_settings,
380             binding_descs,

382             cull_mode,
383         )?;
384         unsafe {
385             device.destroy_pipeline(old_pipeline, None);
386         }
387         Ok(())
388     }

390     /// Cleans up the pipeline resources, destroying the shader modules and pipeline layout.
391     /// # Arguments
392     /// * `device` - The Vulkan logical device to use for destroying the resources.
393     /// # Notes
394     /// * This method should be called when the pipeline is no longer needed, such as during
395     ///   application shutdown or when the pipeline is being recreated.
396     pub fn cleanup(&self, device: &ash::Device) {
397         unsafe {
398             device.destroy_pipeline(self.vk_pipeline, None);
399             device.destroy_pipeline_layout(self.vk_layout, None);
400         }
401     }
