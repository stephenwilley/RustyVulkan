9     fn create_instance(
10         entry: &Entry,
11         event_loop: &ActiveEventLoop,
12         layers: &[*const i8],
13     ) -> Result<Instance, Box<dyn Error>> {
14         let app_name = std::ffi::CString::new("RustyVulkan")?;
15 
16         let app_info = vk::ApplicationInfo {
17             p_application_name: app_name.as_ptr(),
18             application_version: vk::make_api_version(0, 1, 0, 0),
19             p_engine_name: app_name.as_ptr(),
20             engine_version: vk::make_api_version(0, 1, 0, 0),
21             api_version: vk::API_VERSION_1_3,
22             ..Default::default()
23         };
24 
25         let ext_names =
26             enumerate_required_extensions(event_loop.display_handle().unwrap().as_raw())?;
27         let mut extension_ptrs: Vec<*const i8> = ext_names.to_vec();
28 
29         #[cfg(debug_assertions)]
30         extension_ptrs.push(ash::ext::debug_utils::NAME.as_ptr());
31 
32         // Query supported instance extensions
33         let supported_instance_exts =
34             unsafe { entry.enumerate_instance_extension_properties(None)? };
35         let supports_portability_enum = supported_instance_exts.iter().any(|e| {
36             let name = unsafe { std::ffi::CStr::from_ptr(e.extension_name.as_ptr()) };
37             name == vk::KHR_PORTABILITY_ENUMERATION_NAME
38         });
39 
40         // Enable portability enumeration only when the active loader exposes it.
41         // KosmicKrisp does not need it, while MoltenVK and some other drivers do.
42         let mut instance_flags = vk::InstanceCreateFlags::empty();
43         if supports_portability_enum {
44             extension_ptrs.push(vk::KHR_PORTABILITY_ENUMERATION_NAME.as_ptr());
45             instance_flags |= vk::InstanceCreateFlags::ENUMERATE_PORTABILITY_KHR;
46         }
47 
48         // Do NOT push KHR_get_physical_device_properties2 (core since Vulkan 1.1+)
49         let create_info = vk::InstanceCreateInfo {
50             flags: instance_flags,
51             p_application_info: &app_info,
52             enabled_layer_count: layers.len() as u32,
53             pp_enabled_layer_names: layers.as_ptr(),
54             enabled_extension_count: extension_ptrs.len() as u32,
55             pp_enabled_extension_names: extension_ptrs.as_ptr(),
56             ..Default::default()
57         };
58 
59         Ok(unsafe { entry.create_instance(&create_info, None)? })
60     }

81     /// Selects a device and queue family that can both render and present to
82     /// this window.  A discrete GPU is preferred, but only when it meets all
83     /// requirements; the old code picked the first discrete GPU before
84     /// checking presentation support.
85     fn choose_device_and_queue_family(
86         instance: &Instance,
87         physical_devices: &[vk::PhysicalDevice],
88         surface_loader: &surface::Instance,
89         surface: vk::SurfaceKHR,
90     ) -> Result<(vk::PhysicalDevice, u32), Box<dyn Error>> {
91         for prefer_discrete in [true, false] {
92             for &physical_device in physical_devices {
93                 let props = unsafe { instance.get_physical_device_properties(physical_device) };
94                 if (props.device_type == vk::PhysicalDeviceType::DISCRETE_GPU) != prefer_discrete {
95                     continue;
96                 }
97 
98                 // Scene and grass shaders share a 144-byte push-constant ABI.
99                 // Vulkan 1.3 only guarantees 128 bytes, so check before creating resources.
100                 if props.limits.max_push_constants_size
101                     < crate::graphics::pipeline::PUSH_CONSTANT_BYTES
102                 {
103                     eprintln!(
104                         "Skipping GPU: maxPushConstantsSize={} bytes; renderer requires {}",
105                         props.limits.max_push_constants_size,
106                         crate::graphics::pipeline::PUSH_CONSTANT_BYTES,
107                     );
108                     continue;
109                 }
110 
111                 let extensions =
112                     unsafe { instance.enumerate_device_extension_properties(physical_device)? };
113                 let has_extension = |required: &CStr| {
114                     extensions.iter().any(|extension| {
115                         let name = unsafe { CStr::from_ptr(extension.extension_name.as_ptr()) };
116                         name == required
117                     })
118                 };
119                 if !has_extension(vk::KHR_SWAPCHAIN_NAME) {
120                     continue;
121                 }
122 
123                 // Dynamic rendering was promoted to Vulkan 1.3.  Drivers are
124                 // not required to advertise the older KHR extension once they
125                 // expose the core API, so accept either form.
126                 let supports_dynamic_rendering_extension =
127                     has_extension(vk::KHR_DYNAMIC_RENDERING_NAME);
128                 if props.api_version < vk::API_VERSION_1_3 && !supports_dynamic_rendering_extension
129                 {
130                     continue;
131                 }
132 
133                 let supports_dynamic_rendering = unsafe {
134                     let mut features = vk::PhysicalDeviceDynamicRenderingFeatures::default();
135                     let mut features2 =
136                         vk::PhysicalDeviceFeatures2::default().push_next(&mut features);
137                     instance.get_physical_device_features2(physical_device, &mut features2);
138                     features.dynamic_rendering == vk::TRUE
139                 };
140                 if !supports_dynamic_rendering {
141                     continue;
142                 }
143 
144                 let queue_families = unsafe {
145                     instance.get_physical_device_queue_family_properties(physical_device)
146                 };
147                 for (index, family) in queue_families.iter().enumerate() {
148                     if !family.queue_flags.contains(vk::QueueFlags::GRAPHICS) {
149                         continue;
150                     }
151                     if unsafe {
152                         surface_loader.get_physical_device_surface_support(
153                             physical_device,
154                             index as u32,
155                             surface,
156                         )?
157                     } {
158                         println!("🎯 Found graphics/present queue family at index {}", index);
159                         return Ok((physical_device, index as u32));
160                     }
161                 }
162             }
163         }
164 
165         Err("No Vulkan device supports graphics, presentation, dynamic rendering, and at least 144 bytes of push constants".into())
166     }

168     /// Creates a logical device and retrieves the graphics queue.
169     /// # Arguments
170     /// * `instance` - The Vulkan instance.
171     /// * `physical_device` - The physical device to create the logical device from.
172     /// * `queue_family_index` - The index of the graphics queue family.
173     /// # Returns
174     /// * `Result<(ash::Device, vk::Queue), vk::Result>` - A tuple containing the logical device and graphics queue on success, or a Vulkan error on failure.
175     fn create_logical_device_and_queue(
176         instance: &Instance,
177         physical_device: vk::PhysicalDevice,
178         queue_family_index: u32,
179         enable_multi_draw_indirect: bool,
180         enable_draw_indirect_first_instance: bool,
181         enable_wireframe: bool,
182     ) -> Result<(ash::Device, vk::Queue), vk::Result> {
183         let queue_priority = [1.0_f32];
184 
185         // Enable only the features the renderer actually requires.  Copying
186         // all advertised features and forcing anisotropy on made device
187         // creation fail on otherwise usable Vulkan implementations.
188         let device_features = vk::PhysicalDeviceFeatures {
189             // LINE polygon mode requires this feature to be enabled, not just supported.
190             fill_mode_non_solid: if enable_wireframe {
191                 vk::TRUE
192             } else {
193                 vk::FALSE
194             },
195             multi_draw_indirect: if enable_multi_draw_indirect {
196                 vk::TRUE
197             } else {
198                 vk::FALSE
199             },
200             draw_indirect_first_instance: if enable_draw_indirect_first_instance {
201                 vk::TRUE
202             } else {
203                 vk::FALSE
204             },
205             ..Default::default()
206         };
207 
208         let queue_info = vk::DeviceQueueCreateInfo {
209             s_type: vk::StructureType::DEVICE_QUEUE_CREATE_INFO,
210             queue_family_index,
211             queue_count: 1,
212             p_queue_priorities: queue_priority.as_ptr(),
213             ..Default::default()
214         };
215 
216         // Query device extensions on this adapter
217         let supported_dev_exts =
218             unsafe { instance.enumerate_device_extension_properties(physical_device)? };
219 
220         let has_portability_subset = supported_dev_exts.iter().any(|e| {
221             let name = unsafe { std::ffi::CStr::from_ptr(e.extension_name.as_ptr()) };
222             name == vk::KHR_PORTABILITY_SUBSET_NAME
223         });
224 
225         let mut device_extensions: Vec<*const i8> = Vec::new();
226         device_extensions.push(vk::KHR_SWAPCHAIN_NAME.as_ptr());
227         let has_dynamic_rendering_extension = supported_dev_exts.iter().any(|e| {
228             let name = unsafe { std::ffi::CStr::from_ptr(e.extension_name.as_ptr()) };
229             name == vk::KHR_DYNAMIC_RENDERING_NAME
230         });
231         if has_dynamic_rendering_extension {
232             device_extensions.push(vk::KHR_DYNAMIC_RENDERING_NAME.as_ptr());
233         }
234         if has_portability_subset {
235             // Required by portability drivers such as MoltenVK; absent on KosmicKrisp.
236             device_extensions.push(vk::KHR_PORTABILITY_SUBSET_NAME.as_ptr());
237         }
238 
239         let mut dynamic_rendering_features = vk::PhysicalDeviceDynamicRenderingFeatures {
240             dynamic_rendering: vk::TRUE,
241             ..Default::default()
242         };
243 
244         let device_create_info = vk::DeviceCreateInfo {
245             p_next: &mut dynamic_rendering_features as *mut _ as *const _,
246             p_queue_create_infos: &queue_info,
247             queue_create_info_count: 1,
248             pp_enabled_extension_names: device_extensions.as_ptr(),
249             enabled_extension_count: device_extensions.len() as u32,
250             p_enabled_features: &device_features,
251             ..Default::default()
252         };
253 
254         let device = unsafe { instance.create_device(physical_device, &device_create_info, None)? };
255         let queue = unsafe { device.get_device_queue(queue_family_index, 0) };
256         println!("✅ Logical device and graphics queue ready");
257         Ok((device, queue))
258     }

332     /// Create N mapped host-visible uniform buffers sized for `GlobalUbo`.
333     pub(super) fn create_uniform_buffers(
334         allocator: &Allocator,
335         count: usize,
336     ) -> Result<(Vec<vk::Buffer>, Vec<Allocation>), vk::Result> {
337         let mut buffers = Vec::with_capacity(count);
338         let mut allocations = Vec::with_capacity(count);
339 
340         let buffer_size = std::mem::size_of::<GlobalUbo>() as vk::DeviceSize;
341 
342         for _ in 0..count {
343             let buffer_info = vk::BufferCreateInfo {
344                 size: buffer_size,
345                 usage: vk::BufferUsageFlags::UNIFORM_BUFFER,
346                 sharing_mode: vk::SharingMode::EXCLUSIVE,
347                 ..Default::default()
348             };
349             let alloc_info = vk_mem::AllocationCreateInfo {
350                 usage: MemoryUsage::AutoPreferHost,
351                 flags: vk_mem::AllocationCreateFlags::HOST_ACCESS_SEQUENTIAL_WRITE
352                     | vk_mem::AllocationCreateFlags::MAPPED,
353                 ..Default::default()
354             };
355             let (buffer, allocation) =
356                 match unsafe { allocator.create_buffer(&buffer_info, &alloc_info) } {
357                     Ok(resource) => resource,
358                     Err(error) => {
359                         for (buffer, mut allocation) in buffers.into_iter().zip(allocations) {
360                             unsafe { allocator.destroy_buffer(buffer, &mut allocation) };
361                         }
362                         return Err(error);
363                     }
364                 };
365             buffers.push(buffer);
366             allocations.push(allocation);
367         }
368 
369         Ok((buffers, allocations))
370     }

372     /// Build a descriptor pool and one set=0 descriptor set per swapchain image, then write binding 0 to each UBO.
373     pub(super) fn create_set0_descriptor_pool_and_sets(
374         device: &ash::Device,
375         layout: vk::DescriptorSetLayout,
376         ubo_buffers: &[vk::Buffer],
377     ) -> Result<(vk::DescriptorPool, Vec<vk::DescriptorSet>), vk::Result> {
378         let count = ubo_buffers.len() as u32;
379 
380         // Pool: one UBO and one sampler per set
381         let pool_sizes = [
382             vk::DescriptorPoolSize {
383                 ty: vk::DescriptorType::UNIFORM_BUFFER,
384                 descriptor_count: count,
385             },
386             vk::DescriptorPoolSize {
387                 ty: vk::DescriptorType::COMBINED_IMAGE_SAMPLER,
388                 descriptor_count: count,
389             },
390         ];
391         let pool_info = vk::DescriptorPoolCreateInfo {
392             pool_size_count: pool_sizes.len() as u32,
393             p_pool_sizes: pool_sizes.as_ptr(),
394             max_sets: count,
395             ..Default::default()
396         };
397         let pool = unsafe { device.create_descriptor_pool(&pool_info, None)? };
398 
399         // Allocate
400         let layouts = vec![layout; count as usize];
401         let alloc_info = vk::DescriptorSetAllocateInfo {
402             descriptor_pool: pool,
403             descriptor_set_count: count,
404             p_set_layouts: layouts.as_ptr(),
405             ..Default::default()
406         };
407         let sets = match unsafe { device.allocate_descriptor_sets(&alloc_info) } {
408             Ok(sets) => sets,
409             Err(error) => {
410                 unsafe { device.destroy_descriptor_pool(pool, None) };
411                 return Err(error);
412             }
413         };
414 
415         // Write binding 0
416         let range = std::mem::size_of::<GlobalUbo>() as vk::DeviceSize;
417         let mut buf_infos: Vec<vk::DescriptorBufferInfo> = Vec::with_capacity(count as usize);
418         for &b in ubo_buffers {
419             buf_infos.push(vk::DescriptorBufferInfo {
420                 buffer: b,
421                 offset: 0,
422                 range,
423             });
424         }
425 
426         let mut writes: Vec<vk::WriteDescriptorSet> = Vec::with_capacity(count as usize);
427         for i in 0..(count as usize) {
428             writes.push(vk::WriteDescriptorSet {
429                 dst_set: sets[i],
430                 dst_binding: 0,
431                 dst_array_element: 0,
432                 descriptor_count: 1,
433                 descriptor_type: vk::DescriptorType::UNIFORM_BUFFER,
434                 p_buffer_info: &buf_infos[i],
435                 ..Default::default()
436             });
437         }
438         unsafe {
439             device.update_descriptor_sets(&writes, &[]);
440         }
441 
442         Ok((pool, sets))
443     }

445     /// Creates the Vulkan ownership tree stored by `VulkanBase`.
446     pub fn new(window: &Window, event_loop: &ActiveEventLoop) -> Result<Self, Box<dyn Error>> {
447         let engine_settings = EngineSettings {
448             wireframe: false,
449             show_ui: true, // UI enabled by default
450             msaa_samples: 4,
451             // Four 2048² layers use the same texel count as the old 4096² map.
452             shadow_map_resolution: 2048,
453             shadow_distance: 75.0,
454         };
455 
456         let entry = Entry::linked();
457 
458         #[cfg(debug_assertions)]
459         let layer_names = [c"VK_LAYER_KHRONOS_validation"];
460         #[cfg(debug_assertions)]
461         let layer_name_ptrs: Vec<*const i8> = layer_names.iter().map(|s| s.as_ptr()).collect();
462 
463         #[cfg(not(debug_assertions))]
464         let layer_name_ptrs: Vec<*const i8> = Vec::new();
465 
466         #[cfg(debug_assertions)]
467         {
468             unsafe {
469                 let available_layers = entry.enumerate_instance_layer_properties()?;
470                 let validation_layer_name = c"VK_LAYER_KHRONOS_validation";
471                 let is_layer_available = available_layers.iter().any(|layer| {
472                     let name = CStr::from_ptr(layer.layer_name.as_ptr());
473                     name == validation_layer_name
474                 });
475                 if !is_layer_available {
476                     return Err("Validation layers requested, but not available.".into());
477                 }
478             }
479             println!("✅ Validation layers available and requested.");
480         }
481 
482         let instance = Self::create_instance(&entry, event_loop, &layer_name_ptrs)?;
483         println!("🛡️ Vulkan Instance created");
484 
485         #[cfg(debug_assertions)]
486         let (debug_utils_loader, debug_messenger) = Self::setup_debug_messenger(&entry, &instance)?;
487 
488         // Surface capability (including presentation support) is a property
489         // of a physical device, so it must exist before selecting that device.
490         let surface = Self::create_surface(&entry, &instance, window, event_loop)?;
491         let surface_loader = surface::Instance::new(&entry, &instance);
492 
493         let physical_devices = unsafe { instance.enumerate_physical_devices()? };
494 
495         #[cfg(debug_assertions)]
496         Self::print_physical_devices(&instance, &physical_devices);
497 
498         let (physical_device, graphics_queue_family_index) = Self::choose_device_and_queue_family(
499             &instance,
500             &physical_devices,
501             &surface_loader,
502             surface,
503         )?;
504         let chosen_props = unsafe { instance.get_physical_device_properties(physical_device) };
505         let chosen_name = unsafe {
506             CStr::from_ptr(chosen_props.device_name.as_ptr())
507                 .to_str()
508                 .unwrap_or("<invalid utf-8>")
509         };
510         let sample_count_flags_supported = chosen_props.limits.framebuffer_color_sample_counts
511             & chosen_props.limits.framebuffer_depth_sample_counts;
512         let supported_features = unsafe { instance.get_physical_device_features(physical_device) };
513         let supports_wireframe = supported_features.fill_mode_non_solid == vk::TRUE;
514         let supports_multi_draw_indirect = supported_features.multi_draw_indirect == vk::TRUE;
515         let supports_draw_indirect_first_instance =
516             supported_features.draw_indirect_first_instance == vk::TRUE;
517         println!("👉 Selected device for next steps: '{}'", chosen_name);
518 
519         let (device, graphics_queue) = Self::create_logical_device_and_queue(
520             &instance,
521             physical_device,
522             graphics_queue_family_index,
523             supports_multi_draw_indirect,
524             supports_draw_indirect_first_instance,
525             supports_wireframe,
526         )?;
527         // This cache lives for the device lifetime and is shared by every graphics pipeline.
528         let pipeline_cache =
529             unsafe { device.create_pipeline_cache(&vk::PipelineCacheCreateInfo::default(), None)? };
530 
531         let command_pool = Self::create_command_pool(&device, graphics_queue_family_index)?;
532 
533         // Create a Vulkan Memory Allocator (VMA) instance.
534         let mut allocator_info =
535             vk_mem::AllocatorCreateInfo::new(&instance, &device, physical_device);
536         allocator_info.flags |= vk_mem::AllocatorCreateFlags::EXT_MEMORY_BUDGET;
537         let allocator = unsafe { Allocator::new(allocator_info)? };
538 
539         // Binding 1 contains one sampled 2D-array view of all shadow cascades.
540         let ubo_binding = vk::DescriptorSetLayoutBinding {
541             binding: 0,
542             descriptor_type: vk::DescriptorType::UNIFORM_BUFFER,
543             descriptor_count: 1,
544             stage_flags: vk::ShaderStageFlags::VERTEX | vk::ShaderStageFlags::FRAGMENT,
545             p_immutable_samplers: std::ptr::null(),
546             ..Default::default()
547         };
548         let shadow_binding = vk::DescriptorSetLayoutBinding {
549             binding: 1,
550             descriptor_type: vk::DescriptorType::COMBINED_IMAGE_SAMPLER,
551             descriptor_count: 1,
552             stage_flags: vk::ShaderStageFlags::FRAGMENT,
553             p_immutable_samplers: std::ptr::null(),
554             ..Default::default()
555         };
556         let bindings = [ubo_binding, shadow_binding];
557         let set0_info = vk::DescriptorSetLayoutCreateInfo {
558             binding_count: bindings.len() as u32,
559             p_bindings: bindings.as_ptr(),
560             ..Default::default()
561         };
562         let set0_global_layout = unsafe { device.create_descriptor_set_layout(&set0_info, None)? };
563         println!(
564             "🔧 Created global set=0 layout (binding 0 = UBO, binding 1 = cascaded shadow map)"
565         );
566 
567         let swapchain = Swapchain::new(
568             &instance,
569             &device,
570             physical_device,
571             &surface,
572             &surface_loader,
573             window,
574             &allocator,
575             engine_settings.msaa_samples,
576         )?;
577         let swapchain_loader = swapchain::Device::new(&instance, &device);
578 
579         // Create sync objects: N frames-in-flight worth of semaphores/fences
580         let image_count = swapchain.swapchain_image_views.len();
581         let (ubo_buffers, ubo_allocations) = Self::create_uniform_buffers(&allocator, image_count)?;
582         let (set0_descriptor_pool, set0_descriptor_sets) =
583             Self::create_set0_descriptor_pool_and_sets(&device, set0_global_layout, &ubo_buffers)?;
584 
585         // Shadow sampler (no compare; better compatibility with portability subset on macOS)
586         let sampler_info = vk::SamplerCreateInfo {
587             mag_filter: vk::Filter::LINEAR,
588             min_filter: vk::Filter::LINEAR,
589             address_mode_u: vk::SamplerAddressMode::CLAMP_TO_EDGE,
590             address_mode_v: vk::SamplerAddressMode::CLAMP_TO_EDGE,
591             address_mode_w: vk::SamplerAddressMode::CLAMP_TO_EDGE,
592             border_color: vk::BorderColor::FLOAT_OPAQUE_WHITE,
593             unnormalized_coordinates: vk::FALSE,
594             compare_enable: vk::FALSE,
595             ..Default::default()
596         };
597         let shadow_sampler = unsafe { device.create_sampler(&sampler_info, None)? };
598         let mut image_available_semaphores = Vec::with_capacity(INFLIGHT_FRAMES);
599         let mut in_flight_fences = Vec::with_capacity(INFLIGHT_FRAMES);
600         let semaphore_info = vk::SemaphoreCreateInfo::default();
601         let fence_info = vk::FenceCreateInfo {
602             flags: vk::FenceCreateFlags::SIGNALED,
603             ..Default::default()
604         };
605         for _ in 0..INFLIGHT_FRAMES {
606             unsafe {
607                 image_available_semaphores.push(device.create_semaphore(&semaphore_info, None)?);
608                 in_flight_fences.push(device.create_fence(&fence_info, None)?);
609             }
610         }
611 
612         let mut render_finished_semaphores = Vec::with_capacity(image_count);
613         for _ in 0..image_count {
614             unsafe {
615                 render_finished_semaphores.push(device.create_semaphore(&semaphore_info, None)?);
616             }
617         }
618 
619         // Per-swapchain-image tracker: which fence currently owns each image (or null)
620         let image_owner_fence = vec![vk::Fence::null(); image_count];
621 
622         let command_buffers = Self::allocate_command_buffers(
623             &device,
624             command_pool,
625             vk::CommandBufferLevel::PRIMARY,
626             swapchain.swapchain_image_views.len(),
627         )?;
628 
629         let attachment_manager = AttachmentManager::new(image_count);
630 
631         // One query range per swapchain image: frame plus shadow/scene/vegetation boundaries.
632         let query_pool_info = vk::QueryPoolCreateInfo {
633             query_type: vk::QueryType::TIMESTAMP,
634             query_count: (image_count as u32) * TIMESTAMPS_PER_IMAGE,
635             ..Default::default()
636         };
637         let timestamp_query_pool = unsafe { device.create_query_pool(&query_pool_info, None)? };
638 
639         let vulkan_base = Self {
640             instance,
641             physical_device,
642             device,
643             pipeline_cache,
644             allocator: Some(allocator),
645             graphics_queue,
646             // set 0
647             set0_global_layout,
648             set0_descriptor_pool,
649             set0_descriptor_sets,
650             ubo_buffers,
651             ubo_allocations,
652             shadow_sampler,
653             shadow_descriptor_views: vec![vk::ImageView::null(); image_count],
654             timestamp_query_pool,
655             timestamp_period_ns: chosen_props.limits.timestamp_period,
656             last_gpu_timings: GpuPassTimings::default(),
657             last_image_per_slot: vec![None; INFLIGHT_FRAMES],
658             // sync
659             image_available_semaphores,
660             render_finished_semaphores,
661             in_flight_fences,
662             image_owner_fence,
663             // swapchain
664             swapchain,
665             swapchain_loader,
666             surface,
667             surface_loader,
668             // commands
669             command_pool,
670             command_buffers,
671             // misc
672             engine_settings,
673             attachment_manager,
674             #[cfg(debug_assertions)]
675             debug_messenger,
676             #[cfg(debug_assertions)]
677             debug_utils_loader,
678             frame_slot: 0,
679             current_image_index: 0,
680             // RenderGraph also starts at zero; only a later settings or format change
681             // should rebuild the pipelines created during scene setup.
682             pipeline_generation: 0,
683             sample_count_flags_supported,
684             supports_multi_draw_indirect,
685             supports_draw_indirect_first_instance,
686             supports_wireframe,
687             max_draw_indirect_count: chosen_props.limits.max_draw_indirect_count,
688             pending_msaa_samples: None,
689             swapchain_recreation_needed: false,
690         };
691 
692         println!("✅ VulkanBase initialized successfully");
693         Ok(vulkan_base)
694     }
